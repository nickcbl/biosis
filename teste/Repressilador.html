Repressilador (Elowitz & Leibler, 2000)

Modelo adimensional de três genes que se reprimem em ciclo (LacI ⊣ TetR ⊣ λ cI ⊣ LacI).

# Importando as ferramentas necessárias
import time

from scipy.integrate import solve_ivp
from scipy.signal import find_peaks
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation
from IPython.display import HTML

# DEFININDO AS EQUAÇÕES DO REPRESSILADOR (definidas uma única vez)
def repressilator(t, y, alpha, n, beta):
    """
    Sistema de equações diferenciais do Repressilador
    t: tempo
    y: variáveis [m1, m2, m3, p1, p2, p3]
    alpha, n, beta: parâmetros do modelo
    """
    m1, m2, m3, p1, p2, p3 = y

    # Durante os passos de tentativa do solver, p pode ficar levemente negativo;
    # p**n com n não inteiro geraria NaN, então truncamos em zero.
    P1, P2, P3 = max(p1, 0.0), max(p2, 0.0), max(p3, 0.0)

    # Equações para o mRNA (m)
    dm1_dt = -m1 + alpha / (1 + P3**n)   # gene 1 (lacI) reprimido por p3 (cI)
    dm2_dt = -m2 + alpha / (1 + P1**n)   # gene 2 (tetR) reprimido por p1 (LacI)
    dm3_dt = -m3 + alpha / (1 + P2**n)   # gene 3 (cI)   reprimido por p2 (TetR)

    # Equações para as proteínas (p)
    dp1_dt = -beta * (p1 - m1)
    dp2_dt = -beta * (p2 - m2)
    dp3_dt = -beta * (p3 - m3)

    return [dm1_dt, dm2_dt, dm3_dt, dp1_dt, dp2_dt, dp3_dt]

# Tolerâncias do integrador (o padrão rtol=1e-3 é grosseiro para ciclos-limite)
SOLVER_OPTS = dict(method='RK45', rtol=1e-8, atol=1e-10)
# PARÂMETROS DO MODELO (você pode alterar esses valores!)
alpha = 100  # Força de produção
n = 2        # Coeficiente de Hill (cooperatividade)
beta = 0.2   # Razão entre degradação de proteína e mRNA

# Condições iniciais: [m1, m2, m3, p1, p2, p3]
y0 = [0.1, 0.2, 0.3, 0.1, 0.2, 0.3]

# Tempo de simulação
t_span = (0, 200)
t_eval = np.linspace(*t_span, 1000)
# RESOLVENDO AS EQUAÇÕES DIFERENCIAIS
print("Resolvendo as equações do repressilador...")
solution = solve_ivp(repressilator, t_span, y0, args=(alpha, n, beta),
                     t_eval=t_eval, **SOLVER_OPTS)

if not solution.success:
    raise RuntimeError(f"Falha na integração: {solution.message}")

print("Simulação concluída!")
# CRIANDO O GRÁFICO ANIMADO (espaço de fases sendo desenhado)
# Extraindo os resultados
t = solution.t
m1, m2, m3, p1, p2, p3 = solution.y

def com_margem(v, frac=0.05):
    pad = frac * (v.max() - v.min() or 1.0)
    return v.min() - pad, v.max() + pad

fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 8))

# Plot 1: Proteínas ao longo do tempo
for data, label in [(p1, 'Proteína 1 (LacI)'),
                    (p2, 'Proteína 2 (TetR)'),
                    (p3, 'Proteína 3 (λ cI)')]:
    ax1.plot(t, data, label=label, linewidth=2)

ax1.set_xlabel('Tempo')
ax1.set_ylabel('Concentração de Proteína')
ax1.set_title('Repressilador - Dinâmica Temporal')

ax2.set_xlabel('Proteína 1')
ax2.set_ylabel('Proteína 2')
ax2.set_title('Espaço de Fases: Proteína 1 vs Proteína 2')
ax2.grid(True, alpha=0.3)
ax2.set_xlim(*com_margem(p1))
ax2.set_ylim(*com_margem(p2))

step = 4  # desenha 1 a cada 4 pontos (mais fluido)

print("\n=== ANÁLISE DO COMPORTAMENTO OSCILATÓRIO ===")

# Descarta os primeiros 30% da simulação (transiente inicial)

# Picos da Proteína 1; 'prominence' evita contar ruído como oscilação

peaks, _ = find_peaks(p1_ss, prominence=0.1 * variacao) if variacao > 1e-6 else ([], None)
periods = np.diff(t_ss[peaks])

if len(periods) > 0:
    print(f"Número de períodos completos: {len(periods)}")
    print(f"Período médio: {np.mean(periods):.2f} unidades de tempo")
    print(f"Amplitude média dos picos: {np.mean(p1_ss[peaks]):.2f}")
else:
    print("Sistema não está oscilando! Tente aumentar 'alpha'.")

# Dados da simulação
t = solution.t
m1, m2, m3, p1, p2, p3 = solution.y

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5), dpi=60)

# Gráfico 1: Proteínas ao longo do tempo
line1, = ax1.plot([], [], 'b-', label='Proteína 1 (LacI)', linewidth=2)

ax2.set_xlim(*com_margem(p1))
ax2.set_ylim(*com_margem(p2))

frames_2 = range(1, len(t) + 1, 10)

def animate(frame):
    # frame = quantidade de pontos já desenhados (1..len(t))
    line1.set_data(t[:frame], p1[:frame])

    line3.set_data(t[:frame], p3[:frame])

    return line1, line2, line3, phase_line, current_point, time_text

print("Criando animação...")
anim = FuncAnimation(fig, animate, frames=frames_2, init_func=init,

html_anim = anim.to_jshtml()   # renderiza todos os quadros
plt.close(fig)                 # evita mostrar o gráfico estático
HTML(html_anim)
# SIMULAÇÃO EM TEMPO REAL (integra passo a passo e atualiza o gráfico)
# Reaproveita 'repressilator', 'SOLVER_OPTS', alpha, n, beta definidos acima.
dt = 0.5
n_steps = 200

# Estado do zero: não reutiliza y0/variáveis de execuções anteriores
y_current = [0.1, 0.2, 0.3, 0.1, 0.2, 0.3]
t_data = [0.0]
p1_data, p2_data, p3_data = [y_current[3]], [y_current[4]], [y_current[5]]

plt.ion()
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6))

(line1,) = ax1.plot([], [], 'b-', label='Proteína 1', linewidth=2)
(line2,) = ax1.plot([], [], 'r-', label='Proteína 2', linewidth=2)
(line3,) = ax1.plot([], [], 'g-', label='Proteína 3', linewidth=2)
(phase_line,) = ax2.plot([], [], '-', color='purple', linewidth=2, alpha=0.7)

ax1.set_xlabel('Tempo')

ax2.set_xlabel('Proteína 1')
ax2.set_ylabel('Proteína 2')
ax2.set_title('Espaço de Fases - Desenhando Agora!')
ax2.grid(True, alpha=0.3)

plt.tight_layout()

print('Iniciando simulação em tempo real...')

try:
    for i in range(n_steps):
        t_current = i * dt
        t_next = t_current + dt
        sol = solve_ivp(repressilator, [t_current, t_next], y_current,
                        args=(alpha, n, beta), **SOLVER_OPTS)
        if not sol.success:
            raise RuntimeError(sol.message)

        y_current = sol.y[:, -1].tolist()

        # O valor calculado corresponde ao instante t_next (não a t_current)
        t_data.append(t_next)
        p1_data.append(y_current[3])
        p2_data.append(y_current[4])
        p3_data.append(y_current[5])

        # Atualiza gráficos
        line1.set_data(t_data, p1_data)
        line2.set_data(t_data, p2_data)
        line3.set_data(t_data, p3_data)
        phase_line.set_data(p1_data, p2_data)

        # Reajuste dinâmico de limites
        for ax in (ax1, ax2):
            ax.relim()
            ax.autoscale_view()

        fig.canvas.draw_idle()
        fig.canvas.flush_events()

except KeyboardInterrupt:

finally:
    plt.ioff()

def avancar_um_passo():
    """Integra mais um intervalo dt e guarda o resultado (instante t + dt)."""
    t_atual = estado['t'][-1]
    sol = solve_ivp(repressilator, [t_atual, t_atual + dt], estado['y'],
                    args=(alpha, n, beta), **SOLVER_OPTS)
    if not sol.success:
        raise RuntimeError(sol.message)
    estado['y'] = sol.y[:, -1].tolist()
    estado['t'].append(t_atual + dt)
    estado['p1'].append(estado['y'][3])
    estado['p2'].append(estado['y'][4])
    estado['p3'].append(estado['y'][5])

def update(frame):
    # frame = número de passos já integrados (0..n_frames).
    # O FuncAnimation chama o primeiro quadro mais de uma vez; por isso só
    # integramos quando o quadro pedido ainda não foi calculado (idempotente).
    while len(estado['t']) <= frame:
        avancar_um_passo()

    k = frame + 1  # quantidade de pontos a desenhar
    t_d, a, b, c = (estado[x][:k] for x in ('t', 'p1', 'p2', 'p3'))

    line1.set_data(t_d, a)
    line2.set_data(t_d, b)
    line3.set_data(t_d, c)
    phase_line.set_data(a, b)

    # Ajusta eixos dinamicamente
    ax1.set_xlim(0, max(100, t_d[-1]))
    ax1.set_ylim(0, max(max(a), max(b), max(c)) * 1.1)

    def lim(v):
        lo, hi = min(v), max(v)
        pad = 0.1 * (hi - lo) if hi > lo else 1.0
        return lo - pad, hi + pad
    ax2.set_xlim(*lim(a))
    ax2.set_ylim(*lim(b))

    return line1, line2, line3, phase_line

ani = FuncAnimation(fig, update, frames=n_frames, interval=20,
                    repeat=False, blit=False)  # blit=False: eixos mudam a cada quadro

plt.show()
